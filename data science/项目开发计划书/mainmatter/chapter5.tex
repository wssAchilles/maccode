\section{负载预测模型}

\subsection{问题形式化}

能源负载预测问题可形式化为一个监督学习回归任务。给定历史观测数据集：

\begin{equation}
    \mathcal{D} = \{(\mathbf{x}_i, y_i)\}_{i=1}^{N}
\end{equation}

其中，$\mathbf{x}_i \in \mathbb{R}^d$ 为第 $i$ 个样本的 $d$ 维特征向量，$y_i \in \mathbb{R}^+$ 为对应的电力负载值（kWh）。目标是学习映射函数 $f: \mathbb{R}^d \rightarrow \mathbb{R}^+$，使得预测误差最小化：

\begin{equation}
    \min_{f} \frac{1}{N} \sum_{i=1}^{N} \mathcal{L}(y_i, f(\mathbf{x}_i))
\end{equation}

其中 $\mathcal{L}$ 为损失函数，本项目采用均方误差（MSE）。

\subsection{自动模型选择算法}

系统实现了基于时间序列交叉验证的自动模型选择机制，如算法~\ref{alg:auto_select}所示。

\begin{algorithm}[htbp]
    \caption{自动模型选择算法}
    \label{alg:auto_select}
    \begin{algorithmic}[1]
        \Require 训练数据 $\mathcal{D}_{train}$, 候选模型集 $\mathcal{M} = \{RF, LGB, XGB\}$
        \Ensure 最优模型 $m^*$ 及其超参数 $\theta^*$

        \State $best\_score \gets -\infty$
        \State $m^* \gets \text{None}$

        \For{each model $m \in \mathcal{M}$}
        \State $\theta_m \gets \text{RandomizedSearchCV}(m, \mathcal{D}_{train})$ \Comment{超参数搜索}
        \State $score_m \gets \text{TimeSeriesCV}(m, \theta_m, \mathcal{D}_{train})$ \Comment{时间序列CV}

        \If{$score_m > best\_score$}
        \State $best\_score \gets score_m$
        \State $m^* \gets m$
        \State $\theta^* \gets \theta_m$
        \EndIf
        \EndFor

        \State \Return $m^*, \theta^*$
    \end{algorithmic}
\end{algorithm}

\section{储能优化算法}

\subsection{问题建模}

储能调度问题建模为\textbf{混合整数规划}（MIP）问题\cite{wolsey1998integer}，目标是在满足物理约束的前提下，最小化24小时电费支出。

\subsubsection{决策变量}

对于时间窗口 $T = \{0, 1, \ldots, 23\}$，定义以下决策变量：

\begin{table}[htbp]
    \centering
    \caption{MIP决策变量定义}
    \label{tab:mip_vars}
    \begin{tabular}{ccp{6cm}c}
        \toprule
        \textbf{变量}  & \textbf{类型} & \textbf{含义} & \textbf{取值范围}  \\
        \midrule
        $P_{ch}(t)$  & 连续          & $t$ 时刻充电功率  & $[0, P_{max}]$ \\
        $P_{dis}(t)$ & 连续          & $t$ 时刻放电功率  & $[0, P_{max}]$ \\
        $E(t)$       & 连续          & $t$ 时刻电池储能量 & $[0, E_{cap}]$ \\
        $z_{ch}(t)$  & 二元          & $t$ 时刻充电状态  & $\{0, 1\}$     \\
        $z_{dis}(t)$ & 二元          & $t$ 时刻放电状态  & $\{0, 1\}$     \\
        \bottomrule
    \end{tabular}
\end{table}

\subsubsection{目标函数}

最小化净购电成本：

\begin{equation}
    \min \sum_{t \in T} \left[ L(t) + P_{ch}(t) - P_{dis}(t) \right] \cdot \pi(t)
    \label{eq:objective}
\end{equation}

其中，$L(t)$ 为 $t$ 时刻预测负载，$\pi(t)$ 为分时电价。

\subsubsection{约束条件}

\paragraph{约束1: 充放电互斥}
电池不能同时充电和放电：
\begin{equation}
    z_{ch}(t) + z_{dis}(t) \leq 1, \quad \forall t \in T
\end{equation}

\paragraph{约束2: 功率限制}
充放电功率由二元变量控制：
\begin{align}
    P_{ch}(t)  & \leq P_{max} \cdot z_{ch}(t), \quad \forall t \in T  \\
    P_{dis}(t) & \leq P_{max} \cdot z_{dis}(t), \quad \forall t \in T
\end{align}

\paragraph{约束3: 能量守恒}
电池电量动态变化（考虑效率 $\eta = 0.95$）：
\begin{equation}
    E(t) = E(t-1) + \eta \cdot P_{ch}(t) - \frac{P_{dis}(t)}{\eta}, \quad \forall t \in T
\end{equation}

\paragraph{约束4: 容量限制}
\begin{equation}
    0 \leq E(t) \leq E_{cap}, \quad \forall t \in T
\end{equation}

\subsection{求解器配置}

本项目集成Gurobi优化器\cite{gurobi2023}，核心求解流程如算法~\ref{alg:gurobi}所示。

\begin{algorithm}[htbp]
    \caption{储能优化求解算法}
    \label{alg:gurobi}
    \begin{algorithmic}[1]
        \Require 预测负载 $L[0..23]$, 电价 $\pi[0..23]$, 初始SOC $E_0$, 电池参数
        \Ensure 最优调度计划 $\{P_{ch}^*, P_{dis}^*, E^*\}$

        \State $env \gets \text{GurobiEnv}(\text{WLS License})$ \Comment{创建Gurobi环境}
        \State $model \gets env.\text{createModel}(\text{"EnergyOptimization"})$

        \State \Comment{------- 定义决策变量 -------}
        \For{$t \in \{0, 1, \ldots, 23\}$}
        \State $P_{ch}[t] \gets model.\text{addVar}(0, P_{max}, \text{CONTINUOUS})$
        \State $P_{dis}[t] \gets model.\text{addVar}(0, P_{max}, \text{CONTINUOUS})$
        \State $E[t] \gets model.\text{addVar}(0, E_{cap}, \text{CONTINUOUS})$
        \State $z_{ch}[t] \gets model.\text{addVar}(\text{BINARY})$
        \State $z_{dis}[t] \gets model.\text{addVar}(\text{BINARY})$
        \EndFor

        \State \Comment{------- 添加约束 -------}
        \For{$t \in \{0, 1, \ldots, 23\}$}
        \State $model.\text{addConstr}(z_{ch}[t] + z_{dis}[t] \leq 1)$ \Comment{互斥}
        \State $model.\text{addConstr}(P_{ch}[t] \leq P_{max} \cdot z_{ch}[t])$ \Comment{功率}
        \State $model.\text{addConstr}(P_{dis}[t] \leq P_{max} \cdot z_{dis}[t])$
        \State \Comment{能量守恒（略）}
        \EndFor

        \State \Comment{------- 设置目标函数 -------}
        \State $cost \gets \sum_{t} (L[t] + P_{ch}[t] - P_{dis}[t]) \cdot \pi[t]$
        \State $model.\text{setObjective}(cost, \text{MINIMIZE})$

        \State \Comment{------- 求解 -------}
        \State $model.\text{optimize}()$

        \If{$model.\text{status} == \text{OPTIMAL}$}
        \State \Return 提取最优解
        \Else
        \State \textbf{raise} OptimizationError
        \EndIf
    \end{algorithmic}
\end{algorithm}

\section{模型可解释性}

\subsection{SHAP 解释框架}

为增强用户对AI决策的信任，本项目集成了SHAP（SHapley Additive exPlanations）框架\cite{lundberg2017shap}，提供特征级别的预测解释。

SHAP值的计算基于博弈论中的Shapley值，对于特征 $i$，其SHAP值定义为：

\begin{equation}
    \phi_i = \sum_{S \subseteq F \setminus \{i\}} \frac{|S|!(|F|-|S|-1)!}{|F|!} \left[ f_{S \cup \{i\}}(x_{S \cup \{i\}}) - f_S(x_S) \right]
\end{equation}

其中 $F$ 为全部特征集合，$f_S$ 为仅使用特征子集 $S$ 的预测值。
