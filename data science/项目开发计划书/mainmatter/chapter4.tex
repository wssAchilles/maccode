\subsubsection{FR-01: 用户身份认证}

\begin{table}[H]
  \centering
  \caption{FR-01 用户身份认证需求}
  \begin{tabular}{p{3cm}p{10cm}}
    \toprule
    \textbf{需求编号} & FR-01                                        \\
    \textbf{需求名称} & 用户身份认证                                       \\
    \textbf{优先级}  & 高                                            \\
    \textbf{描述}   & 系统应支持用户通过Firebase Authentication进行注册、登录和登出操作 \\

    \end{enumerate}                        \\
    \bottomrule
  \end{tabular}
\end{table}

\subsubsection{FR-02: 能源负载预测}

\begin{table}[htbp]
  \centering
  \caption{FR-02 能源负载预测需求}
  \begin{tabular}{p{3cm}p{10cm}}
    \toprule
    \textbf{需求编号} & FR-02                 \\
    \textbf{需求名称} & 能源负载预测                \\
    \textbf{优先级}  & 高                     \\
    \textbf{描述}   & 系统应提供未来24小时的电力负载预测功能  \\
    \textbf{输入}   & 目标日期、温度调整参数（可选）       \\
    \textbf{输出}   & 24个时间点的预测负载值（kWh）     \\
    \textbf{验收标准} &
    \begin{enumerate}[leftmargin=*, nosep]
      \item 预测精度：$R^2 \geq 0.85$
      \item 响应时间 $\leq 3$ 秒
      \item 支持温度偏移调整
    \end{enumerate} \\
    \bottomrule
  \end{tabular}
\end{table}

\subsection{核心用例描述}

\subsubsection{UC-03: 运行储能优化}

\begin{table}[H]
  \centering
  \caption{UC-03 运行储能优化用例描述}
  \begin{tabular}{p{3cm}p{10cm}}
    \toprule
    \textbf{用例编号} & UC-03                 \\
    \textbf{用例名称} & 运行储能优化                \\
    \textbf{参与者}  & 普通用户                  \\
    \textbf{前置条件} & 用户已登录系统               \\
    \textbf{后置条件} & 生成24小时最优充放电计划         \\
    \midrule
    \textbf{主流程}  &
    \begin{enumerate}[leftmargin=*, nosep]
      \item 用户进入优化仪表盘页面
      \item 用户设置优化参数（初始SOC、电池容量等）
      \item 用户选择目标日期
      \item 用户点击"运行优化"按钮
      \item 系统调用负载预测服务获取预测值
      \item 系统调用优化服务求解最优调度
      \item 系统返回并展示优化结果
    \end{enumerate} \\
    \midrule
    \textbf{异常流程} &
    \begin{itemize}[leftmargin=*, nosep]
      \item E1: 若预测服务失败，显示错误提示，建议稍后重试
      \item E2: 若优化无解，显示"无可行解"提示
      \item E3: 若优化超时，显示超时错误
    \end{itemize}   \\
    \bottomrule
  \end{tabular}
\end{table}

\begin{table}[htbp]
  \centering
  \caption{数据存储设计}
  \label{tab:storage_design}
  \begin{tabular}{lp{3cm}p{4cm}p{4cm}}
    \toprule
    \textbf{存储类型} & \textbf{技术选型}   & \textbf{存储内容}  & \textbf{选型理由} \\
    \midrule
    文档数据库         & Cloud Firestore & 用户信息、操作历史、任务状态 & 实时同步、无需Schema \\
    对象存储          & Cloud Storage   & CSV数据文件、ML模型文件 & 大文件存储、高可用     \\

  \end{tabular}
